router.get("/login", (req, res) => {
  if (hasTvboxSession(req)) return res.redirect("/tvbox");
  if (!PASSWORD)
    return res
      .status(503)
      .send(
        loginPage(
          "Configure TVBOX_PASSWORD no servidor antes de usar o painel.",
        ),
      );
  res.send(
    loginPage(
      req.query.error === "1" ? "Senha inválida. Tente novamente." : "",
    ),
  );
});

router.post("/login", (req, res) => {
  if (!PASSWORD)
    return res
      .status(503)
      .send(loginPage("Configuração de segurança ausente no servidor."));
  if (!passwordMatches(req.body.password))
    return res.redirect("/tvbox/login?error=1");
  const expiresAt = Date.now() + SESSION_DURATION_MS;
  res.cookie(SESSION_COOKIE, signSession({ expiresAt }), {
    httpOnly: true,
    sameSite: "lax",
    secure: req.secure,
    path: "/tvbox",
    maxAge: SESSION_DURATION_MS,
  });
  res.redirect("/tvbox");
});

router.post("/logout", (req, res) => {

  res.redirect("/tvbox/login");
});

router.get("/api/channels", (req, res) => {
  const dbData = loadDB();
  const categoriesMeta = loadCategoriesMeta();
  const forwardedProto = String(req.headers["x-forwarded-proto"] || "").split(",")[0];
  const baseUrl = `${forwardedProto || req.protocol}://${req.get("host")}`;

  const cleanData = dbData.map(({ id, name, url, type, category, categories, poster, source }) => {
    const categoryKey = String(category || "outros").trim();
    const categoryPoster = categoriesMeta[categoryKey]?.poster || null;

    const proxyUrl = source === "embedtv-channels"
      ? `${baseUrl}/tvbox/api/live/${encodeURIComponent(id)}.m3u8`
      : url;
    return {
      id,
      name,
      url: proxyUrl,
      type,
      category,
      categories: itemCategories({ category, categories }),
      poster: poster || categoryPoster,
    };
  });

  res.setHeader("Access-Control-Allow-Origin", "*");
  res.json(cleanData);
});

function bannerCatalog() {
  return [
    ...loadDB().map(item => ({ key: "item:" + item.id, name: item.name })),
    ...loadSeriesCatalog().map(item => ({ key: "series:" + item.id, name: item.title || item.name || item.id }))

}

router.post("/home/banners", (req, res) => {
  const list = value => Array.isArray(value) ? value : value == null ? [] : [value];
  const targets = list(req.body.bannerTarget), titles = list(req.body.bannerTitle);
  const subtitles = list(req.body.bannerSubtitle), images = list(req.body.bannerImage), enabled = list(req.body.bannerEnabled);
  const known = new Set(bannerCatalog().map(item => item.key));
  const banners = targets.map((target, index) => ({
    target: String(target), title: String(titles[index] || "").trim().slice(0, 160),
    subtitle: String(subtitles[index] || "").trim().slice(0, 320),
    image: /^https?:\/\//i.test(String(images[index] || "")) ? String(images[index]).trim() : "",
    enabled: enabled[index] !== "false"
  })).filter(banner => known.has(banner.target)).slice(0, 12);
  saveHomeConfig({ ...loadHomeConfig(), banners });
  res.redirect("/tvbox/home?saved=1");
});

router.post("/posters/fill", async (req, res) => {
  const dbData = loadDB();
  let updated = 0;
  for (const item of dbData) {
    if (item.poster) continue;
    const result = await findPoster(item.name, "movie");
    if (result?.poster) {
      item.poster = result.poster;
      updated++;
    }
  }
  saveDB(dbData);
  res.redirect(`/tvbox?covers=${updated}`);
});

router.post("/series/posters/fill", async (req, res) => {
  const catalog = loadSeriesCatalog();
  let updated = 0;
  for (const item of catalog) {
    if (item.poster) continue;
    const result = await findPoster(item.title || item.name || item.id, "tv");
    if (!result?.poster) continue;
    item.poster = result.poster;
    const file = path.join(SERIES_DIRECTORY, `${item.id}.json`);
    if (fs.existsSync(file)) {
      const details = JSON.parse(fs.readFileSync(file, "utf-8"));
      details.poster = result.poster;
      fs.writeFileSync(file, JSON.stringify(details, null, 2), "utf-8");
    }
    updated++;
  }
  saveSeriesCatalog(catalog);
  res.redirect(`/tvbox/series?covers=${updated}`);
});

router.post("/series/add", (req, res) => {
  const id = String(req.body.id || "").trim();
  const title = String(req.body.title || "").trim();
  const poster = String(req.body.poster || "").trim();
  if (!title || !/^[A-Za-z0-9_-]+$/.test(id))
    return res
      .status(400)
      .send(
        tvboxPage(
          "Erro",
          '<section class="panel"><h1>Dados inválidos</h1><p class="muted">Use um título e um ID válido.</p></section>',
        ),
      );
  const catalog = loadSeriesCatalog();
  const file = path.join(SERIES_DIRECTORY, `${id}.json`);
  if (catalog.some((item) => item.id === id) || fs.existsSync(file))
    return res
      .status(409)
      .send(
        tvboxPage(
          "Erro",
          `<section class="panel"><h1>Essa série já existe</h1><a class="btn" href="/tvbox/series">Voltar</a></section>`,
        ),
      );
  fs.mkdirSync(SERIES_DIRECTORY, { recursive: true });
  const validPoster = /^https?:\/\//i.test(poster) ? poster : null;
  catalog.push({
    id,
    title,
    endpoint: `series/${id}.json`,
    poster: validPoster,
  });
  saveSeriesCatalog(catalog);
  fs.writeFileSync(
    file,
    JSON.stringify({ title, poster: validPoster, seasons: [] }, null, 2),
    "utf-8",
  );
  res.redirect(`/tvbox/series/${encodeURIComponent(id)}`);
});

        tvboxPage(

          '<section class="panel"><h1>Arquivo ausente</h1><p class="muted">Selecione o JSON exportado pelo catálogo M3U.</p></section>',

  let payload;

  } catch {

      );
  }
  const incoming = Array.isArray(payload) ? payload : payload.channels;
  if (!Array.isArray(incoming))
    return res
      .status(400)
      .send(
        tvboxPage(
          "Importar M3U",
          '<section class="panel"><h1>Formato inválido</h1><p class="muted">Use o arquivo gerado pelo catálogo M3U.</p></section>',
        ),
      );
  if (req.body.target === "category") return importM3UCategory(req, res, incoming);

  if (req.body.target === "episodes") {
    const seriesId = String(req.body.seriesId || "");
    const seasonNumber = Number(req.body.seasonNumber);
    const startEpisode = Number(req.body.startEpisode) || 1;
    const seriesFile = path.join(SERIES_DIRECTORY, `${seriesId}.json`);
    if (
      !/^[A-Za-z0-9_-]+$/.test(seriesId) ||
      !fs.existsSync(seriesFile) ||
      !Number.isInteger(seasonNumber) ||
      seasonNumber < 1
    )
      return res
        .status(400)
        .send(
          tvboxPage(
            "Importar M3U",

          ),
        );
    const seriesData = JSON.parse(fs.readFileSync(seriesFile, "utf-8"));
    seriesData.seasons ||= [];
    let season = seriesData.seasons.find(
      (item) => Number(item.number) === seasonNumber,
    );

      season = { number: seasonNumber, episodes: [] };

    }
    season.episodes ||= [];
    const knownUrls = new Set(season.episodes.map((item) => item.url));
    let added = 0,
      ignored = 0,
      nextNumber = Math.max(
        startEpisode,
        ...season.episodes.map((item) => Number(item.number) + 1),
        1,
      );
    for (const item of incoming) {
      const url = String(item?.url || "").trim();
      const title = String(item?.name || item?.title || "").trim();
      if (!title || !/^https?:\/\//i.test(url) || knownUrls.has(url)) {
        ignored++;
        continue;
      }
      season.episodes.push({
        number: nextNumber++,
        title,
        url,
        type: item.type === "mp4" ? "mp4" : "hls",
      });
      knownUrls.add(url);
      added++;
    }
    fs.writeFileSync(seriesFile, JSON.stringify(seriesData, null, 2), "utf-8");
    return res.send(
      tvboxPage(
        "Importação concluída",

function importM3UCategory(req, res, incoming) {
  const category = String(req.body.category || "").trim();
  if (!category) return res.status(400).send("Categoria ausente");
  const normalize = value => String(value || "").normalize("NFD").replace(/[\u0300-\u036f]/g, "").trim().toLocaleLowerCase("pt-BR").replace(/\s+/g, " ");
  const db = loadDB();
  let added = 0, updated = 0, ignored = 0;
  const used = new Set(db.map(item => String(item.id)));
  let nextId = Math.max(Date.now(), ...db.map(item => Number(item.id) || 0)) + 1;
  for (const raw of incoming) {
    const name = String(raw?.name || raw?.title || "").trim();
    const url = String(raw?.url || "").trim();
    if (!name || !/^https?:\/\//i.test(url) || !["http:", "https:"].includes(new URL(url).protocol)) { ignored++; continue; }
    const matches = db.filter(item => hasCategory(item, category) && normalize(item.name) === normalize(name));
    if (matches.length > 1) { ignored++; continue; }
    if (matches.length === 1) {
      if (matches[0].url !== url || matches[0].type !== "mp4") {
        matches[0].url = url;
        matches[0].type = "mp4";
        updated++;
      } else ignored++;
      continue;
    }
    const poster = String(raw.poster || raw.logo || "").trim();
    while (used.has(String(nextId))) nextId++;
    const id = nextId++;
    used.add(String(id));
    db.push({ id, name, url, type: "mp4", category, categories: [category], ...(/^https?:\/\//i.test(poster) ? { poster } : {}) });
    added++;
  }
  if (added || updated) saveDB(db);
  res.send(tvboxPage("Importação concluída", `<section class="panel"><h1>Importação concluída</h1><p>${updated} atualizados · ${added} adicionados · ${ignored} ignorados</p><a class="btn primary" href="/tvbox/categories">Voltar às categorias</a></section>`));
}

  }

              // Usa o NAME cadastrado no painel,

              if (!name) {
                failed++;
                console.warn("[KICK] Canal sem nome:", channel.id);
                return null;
              }

              const kickName = name
                .toLowerCase()
                .normalize("NFD")
                .replace(/[\u0300-\u036f]/g, "")
                .replace(/^@/, "")
                .replace(/\s+/g, "");

              const response = await fetch(
                `https://kick.com/api/v2/channels/${encodeURIComponent(kickName)}`,
                {
                  headers: {
                    Accept: "application/json",
                    "User-Agent": "Mozilla/5.0",
                    Referer: "https://kick.com/",
                  },
                  signal: AbortSignal.timeout(10000),
                },
              );

              if (!response.ok) {
                failed++;

                console.warn(`[KICK] ${kickName}: HTTP ${response.status}`);

                return null;

              return {
                id: channel.id,
                name: channel.name || data.user?.username || kickName,
                kickName: data.slug || kickName,

                url: data.playback_url || null,
                playbackUrl: data.playback_url || null,

function isPrivateIp(ip) {
  const family = net.isIP(ip);

  if (family === 4) {
    const p = ip.split(".").map(Number);

    return (
      p[0] === 0 ||
      p[0] === 10 ||
      p[0] === 127 ||
      p[0] >= 224 ||
      (p[0] === 100 && p[1] >= 64 && p[1] <= 127) ||
      (p[0] === 169 && p[1] === 254) ||
      (p[0] === 172 && p[1] >= 16 && p[1] <= 31) ||
      (p[0] === 192 && p[1] === 168) ||
      (p[0] === 192 && p[1] === 0 && p[2] === 0) ||
      (p[0] === 192 && p[1] === 0 && p[2] === 2) ||
      (p[0] === 198 && p[1] >= 18 && p[1] <= 19) ||
      (p[0] === 198 && p[1] === 51 && p[2] === 100) ||
      (p[0] === 203 && p[1] === 0 && p[2] === 113)
    );
  }

  if (family === 6) {
    const normalized = ip.toLowerCase();

    if (
      normalized === "::" ||
      normalized === "::1" ||
      normalized.startsWith("fc") ||
      normalized.startsWith("fd") ||
      normalized.startsWith("fe8") ||
      normalized.startsWith("fe9") ||
      normalized.startsWith("fea") ||
      normalized.startsWith("feb") ||
      normalized.startsWith("ff")
    ) {
      return true;
    }

    if (normalized.startsWith("::ffff:")) {
      const mapped = normalized.slice(7);

      if (net.isIP(mapped) === 4) {
        return isPrivateIp(mapped);
      }
    }
  }

  return false;
}

async function validateHlsUrl(rawUrl) {
  const target = new URL(rawUrl);

  if (
    !["http:", "https:"].includes(target.protocol) ||
    target.username ||
    target.password ||
    !target.hostname
  ) {
    throw new Error("URL HLS inválida");
  }

  const hostname = target.hostname.toLowerCase();

  if (
    hostname === "localhost" ||
    hostname.endsWith(".localhost") ||
    hostname.endsWith(".local") ||
    hostname.endsWith(".internal")
  ) {
    throw new Error("Destino interno não permitido");
  }

  if (net.isIP(hostname)) {
    if (isPrivateIp(hostname)) {
      throw new Error("IP privado não permitido");
    }
  } else {
    const addresses = await dns.lookup(hostname, {
      all: true,
    });

    if (
      !addresses.length ||
      addresses.some((entry) => isPrivateIp(entry.address))
    ) {
      throw new Error("Destino não permitido");
    }
  }

  return target;
}

const proxySessions = new Map();
const PROXY_SESSION_IDLE_MS = 15 * 60_000;

setInterval(() => {
  const now = Date.now();
  for (const [id, session] of proxySessions) {
    if (now - session.lastAccess > PROXY_SESSION_IDLE_MS) {
      proxySessions.delete(id);
    }
  }
}, 60_000).unref?.();

function getProxySession(id) {
  const validId = /^[a-f0-9]{24}$/i.test(String(id || "")) ? String(id) : null;
  let session = validId ? proxySessions.get(validId) : null;
  if (!session) {
    session = {
      id: crypto.randomBytes(12).toString("hex"),
      cookies: new Map(),
      lastAccess: Date.now(),
    };
    proxySessions.set(session.id, session);
  }
  session.lastAccess = Date.now();
  return session;
}

function cookieHeader(session, target) {
  if (!session) return "";
  const cookies = session.cookies.get(target.hostname.toLowerCase());
  return cookies ? [...cookies.values()].join("; ") : "";
}

function saveCookies(session, target, setCookie) {
  if (!session || !setCookie) return;
  const values = Array.isArray(setCookie) ? setCookie : [setCookie];
  const host = target.hostname.toLowerCase();
  const jar = session.cookies.get(host) || new Map();
  for (const value of values) {
    const pair = String(value).split(";", 1)[0];
    const name = pair.split("=", 1)[0]?.trim();
    if (name) jar.set(name, pair);
  }
  if (jar.size) session.cookies.set(host, jar);
}

function buildProxyUrl(url, session) {
  return (
    `${HLS_PROXY_PATH}?url=` +
    encodeURIComponent(url) +
    `&session=${encodeURIComponent(session.id)}`
  );
}

function rewritePlaylist(content, base, session) {
  return content
    .split(/\r?\n/)
    .map((line) => {
      const trimmed = line.trim();

      if (!trimmed) {
        return line;
      }

      // Tags com URI, como EXT-X-KEY,
      // EXT-X-MAP, EXT-X-PART e EXT-X-MEDIA.
      if (trimmed.startsWith("#")) {
        return line.replace(
          /URI="([^"]+)"/g,
          (_, uri) => {
            const absolute = new URL(
              uri,
              base
            ).href;

            return (
              `URI="${buildProxyUrl(absolute, session)}"`
            );
          }
        );
      }

      // Segmentos e playlists filhas.
      const absolute = new URL(
        trimmed,
        base
      ).href;

      return buildProxyUrl(absolute, session);
    })
    .join("\n");
}

function wait(ms) {
  return new Promise((resolve) => setTimeout(resolve, ms));
}

function retryableHlsStatus(status) {
  return [408, 425, 429, 500, 502, 503, 504].includes(status);
}

async function fetchThroughWarpOnce(url, range, session) {
  let currentUrl = url;

  for (let i = 0; i <= MAX_REDIRECTS; i++) {
    const target = await validateHlsUrl(
      currentUrl
    );

    const response = await axios.get(
      target.href,
      {
        httpAgent: warpAgent,
        httpsAgent: warpAgent,
        proxy: false,
        responseType: "stream",
        timeout: 20000,
        maxRedirects: 0,
        validateStatus: () => true,
        headers: {
          "User-Agent": "Mozilla/5.0",
          Accept: "*/*",
          ...(cookieHeader(session, target)
            ? { Cookie: cookieHeader(session, target) }
            : {}),
          ...(range
            ? { Range: range }
            : {}),
        },
      }
    );

    if (
      [301, 302, 303, 307, 308].includes(
        response.status
      )
    ) {
      const location =
        response.headers.location;

      response.data.destroy();

      if (!location) {
        throw new Error(
          "Redirecionamento sem Location"
        );
      }

      currentUrl = new URL(
        location,
        target
      ).href;

      continue;
    }

    saveCookies(session, target, response.headers["set-cookie"]);
    return {
      response,
      finalUrl: target,
    };
  }

  throw new Error(
    "Muitos redirecionamentos"
  );
}

// Uma falha transitória no WARP/servidor de origem não deve encerrar a live
// imediatamente. Segmentos e playlists recebem duas novas tentativas curtas.
async function fetchThroughWarp(url, range, session) {
  let lastError = null;

  for (let attempt = 0; attempt < 3; attempt++) {
    try {
      const result = await fetchThroughWarpOnce(url, range, session);
      if (!retryableHlsStatus(result.response.status) || attempt === 2) {
        return result;
      }
      result.response.data.destroy();
      await wait(350 * (attempt + 1));
    } catch (error) {
      lastError = error;
      if (attempt === 2) throw error;
      await wait(350 * (attempt + 1));
    }
  }

  throw lastError || new Error("Falha temporária ao buscar HLS");
}

// Cache por playlist ativa.
// O servidor acompanha apenas canais recentemente solicitados.

const livePlaylists = new Map();

const PLAYLIST_IDLE_MS = 10 * 60_000;
const PLAYLIST_MIN_POLL_MS = 2_000;
const PLAYLIST_MAX_POLL_MS = 10_000;
// Não entregue uma playlist velha por muito tempo: depois de 30 s o app
// recebe erro, baixa novamente o catálogo e pode usar uma URL assinada nova.
const PLAYLIST_MAX_STALE_MS = 30_000;

function playlistPollMs(body) {
  const match = body.match(
    /^#EXT-X-TARGETDURATION:\s*(\d+)/m
  );

  const seconds = match
    ? Number(match[1])
    : 6;

  return Math.max(
    PLAYLIST_MIN_POLL_MS,
    Math.min(
      PLAYLIST_MAX_POLL_MS,
      Math.floor(seconds * 500)
    )
  );
}

function playlistDetails(body) {
  const seq = body.match(
    /^#EXT-X-MEDIA-SEQUENCE:\s*(\d+)/m
  );

  return {
    sequence: seq ? Number(seq[1]) : null,
    ended: /^#EXT-X-ENDLIST\s*$/m.test(body),
    master: /^#EXT-X-STREAM-INF:/m.test(body),
  };
}

async function readPlaylist(url, session) {
  const { response, finalUrl } =
    await fetchThroughWarp(url, undefined, session);

  if (response.status !== 200) {
    response.data.destroy();

    throw new Error(
      `Playlist origem HTTP ${response.status}`
    );
  }

  const chunks = [];
  let total = 0;

  for await (const chunk of response.data) {
    total += chunk.length;

    if (total > MAX_PLAYLIST_SIZE) {
      response.data.destroy();
      throw new Error("Playlist muito grande");
    }

    chunks.push(chunk);
  }

  const body = Buffer.concat(chunks)
    .toString("utf8");

  if (!body.trimStart().startsWith("#EXTM3U")) {
    throw new Error(
      "Resposta da origem não é playlist HLS válida"
    );
  }

  return {
    body,
    finalUrl: finalUrl.href,
  };
}

function stopPlaylistPoll(entry) {
  if (entry.timer) {
    clearTimeout(entry.timer);
  }

  entry.timer = null;
  livePlaylists.delete(entry.url);
}

function schedulePlaylistPoll(entry) {
  if (entry.timer) {
    clearTimeout(entry.timer);
  }

  if (
    Date.now() - entry.lastClientAt > PLAYLIST_IDLE_MS ||
    entry.ended ||
    entry.master
  ) {
    stopPlaylistPoll(entry);
    return;
  }

  entry.timer = setTimeout(async () => {
    entry.timer = null;

    if (
      Date.now() - entry.lastClientAt >
      PLAYLIST_IDLE_MS
    ) {
      stopPlaylistPoll(entry);
      return;
    }

    try {
      await updatePlaylist(entry);
    } catch (error) {
      console.warn(
        "[HLS POLL] Falha:",
        error.message,
        entry.url
      );
    }

    if (livePlaylists.get(entry.url) === entry) {
      schedulePlaylistPoll(entry);
    }
  }, entry.pollMs);

  entry.timer.unref?.();
}

async function updatePlaylist(entry) {
  if (entry.pending) {
    return entry.pending;
  }

  entry.pending = (async () => {
    const { body, finalUrl } =
      await readPlaylist(entry.url, entry.session);

    const details = playlistDetails(body);
    const previousSeq = entry.sequence;

    entry.body = rewritePlaylist(body, finalUrl, entry.session);

    entry.updatedAt = Date.now();
    entry.pollMs = playlistPollMs(body);
    entry.sequence = details.sequence;
    entry.ended = details.ended;
    entry.master = details.master;

    if (previousSeq !== entry.sequence) {
      console.log(
        `[HLS POLL] seq ${previousSeq} -> ${entry.sequence}; intervalo=${entry.pollMs}ms`
      );
    }

    if (entry.ended) {
      console.warn(
        "[HLS POLL] A ORIGEM enviou #EXT-X-ENDLIST:",
        entry.url
      );
    }

    return entry.body;
  })();

  try {
    return await entry.pending;
  } finally {
    entry.pending = null;
  }
}

async function getPlaylist(url, requestedSessionId) {
  let entry = livePlaylists.get(url);

  if (!entry) {
    entry = {
      url,
      session: getProxySession(requestedSessionId),
      body: null,
      pending: null,
      timer: null,
      updatedAt: 0,
      lastClientAt: Date.now(),
      pollMs: 3_000,
      sequence: null,
      ended: false,
      master: false,
    };

    livePlaylists.set(url, entry);
  }

  entry.session.lastAccess = Date.now();
  entry.lastClientAt = Date.now();

  if (!entry.body) {
    try {
      await updatePlaylist(entry);
    } catch (error) {
      stopPlaylistPoll(entry);
      throw error;
    }
  } else if (
    Date.now() - entry.updatedAt >
    entry.pollMs * 2
  ) {
    // Se a atualização atrasar, uma requisição
    // do player também pode renová-la.

    try {
      await updatePlaylist(entry);
    } catch (error) {
      if (
        Date.now() - entry.updatedAt >
        PLAYLIST_MAX_STALE_MS
      ) {
        throw error;
      }
    }
  }

  if (
    !entry.timer &&
    !entry.ended &&
    !entry.master
  ) {
    schedulePlaylistPoll(entry);
  }

  return entry.body;
}

      const urlPath = new URL(rawUrl)
        .pathname.toLowerCase();

      // PLAYLISTS M3U8 COM POLLING
      // ==================================

        const body = await getPlaylist(rawUrl, req.query.session);

          .status(200)
          .type("application/vnd.apple.mpegurl")
          .send(body);

      // SEGMENTOS E OUTROS RECURSOS
      // ==================================

      // Segmentos TS, M4S, MP4 fragmentado,
      // chaves e URLs sem extensão:
      // mantidos sob demanda.

      const { response, finalUrl } =
        await fetchThroughWarp(
          rawUrl,
          req.headers.range,
          session
        );

      if (
        response.status !== 200 &&
        response.status !== 206
      ) {
        response.data.destroy();

        return res.status(502).json({
          error:
            `Origem retornou HTTP ${response.status}`,
        });
      }

      const contentType =
        response.headers["content-type"] || "";

      const isPlaylist = /mpegurl/i.test(
        contentType
      );

      // ==================================
      // PLAYLISTS SEM EXTENSÃO .M3U8
      // ==================================

      if (isPlaylist) {
        const chunks = [];
        let total = 0;

        for await (
          const chunk of response.data
        ) {
          total += chunk.length;

          if (total > MAX_PLAYLIST_SIZE) {
            response.data.destroy();

            return res.status(502).json({
              error: "Playlist muito grande",
            });
          }

          chunks.push(chunk);
        }

        const body = Buffer.concat(chunks)
          .toString("utf8");

        if (
          !body.trimStart()
            .startsWith("#EXTM3U")
        ) {
          return res.status(502).json({
            error: "Playlist HLS inválida",
          });
        }

        return res
          .status(200)
          .type("application/vnd.apple.mpegurl")
          .send(
            rewritePlaylist(body, finalUrl, session)
          );
      }

      // ==================================
      // STREAM DOS SEGMENTOS BINÁRIOS
      // ==================================

      res.status(response.status);

      res.setHeader(
        "Content-Type",
        contentType ||
          "application/octet-stream"
      );

      for (const header of [
        "content-length",
        "content-range",
        "accept-ranges",
      ]) {
        if (response.headers[header]) {
          res.setHeader(
            header,
            response.headers[header]
          );
        }
      }

      response.data.on(
        "error",
        (error) => {
          console.error(
            "[HLS PROXY] Erro no segmento:",
            error.message
          );

          if (!res.destroyed) {
            res.destroy(error);
          }
        }
      );

      res.on("close", () => {
        if (!res.writableFinished) {
          response.data.destroy();
        }
      });

      return response.data.pipe(res);
    } catch (error) {
      console.error(
        "[HLS PROXY] Erro:",
        error.message
      );

      if (res.headersSent) {
        return res.destroy(error);
      }

      return res.status(502).json({
        error:
          "Falha ao acessar HLS pelo WARP",
        detail: error.message,
      });
    }
  }
);

// Mantém category como principal para clientes antigos e categories para múltiplas.
function itemCategories(item) {
  const values = [item.category, ...(Array.isArray(item.categories) ? item.categories : [])];
  const seen = new Set();
  return values.map(value => String(value || "").trim()).filter(value => {
    const key = value.toLocaleLowerCase("pt-BR");
    if (!value || seen.has(key)) return false;
    seen.add(key);
    return true;
  });
}

function categoriesFromForm(body) {
  const values = Array.isArray(body.categories) ? body.categories : [body.categories];
  const categories = itemCategories({
    category: body.category,
    categories: [...values, String(body.newCategories || "").split(";")].flat()
  });
  return categories.length ? categories : ["outros"];
}

function allCategoryNames(items, meta = {}) {
  return [...new Set(["filmes", "futebol", "kick", ...Object.keys(meta), ...items.flatMap(itemCategories)])]
    .sort((a, b) => a.localeCompare(b, "pt-BR"));
}

function hasCategory(item, name) {
  return itemCategories(item).some(value => value.toLocaleLowerCase("pt-BR") === String(name).toLocaleLowerCase("pt-BR"));
}

module.exports = { itemCategories, categoriesFromForm, allCategoryNames, hasCategory };

async function syncChannels(force = false) {
  const payload = await loadChannels(force);
  const categoryById = new Map(payload.categories.map((category) => [Number(category.id), String(category.name || "outros").trim()]));
  const imported = payload.channels.map((channel) => {
    const category = (Array.isArray(channel.categories) ? channel.categories : [])
      .map((id) => categoryById.get(Number(id)))
      .find((name) => name && name.toLowerCase() !== "todos") || "EmbedTV";
    const id = `embedtv-channel-${hls.slug(channel.id || channel.name)}`;
    return {
      id,
      name: channel.name || channel.id || id,
      // O banco público guarda apenas o endereço do proxy. A origem fica
      // separada para o serviço HLS continuar renovando o token do EmbedTV.
      url: `${BASE_URL}/tvbox/api/live/${encodeURIComponent(id)}.m3u8`,
      originUrl: channel.url,
      type: "hls",
      category,
      poster: channel.image || channel.preview || null,
      source: "embedtv-channels",
      sourceId: String(channel.id || ""),
    };
  }).filter((channel) => channel.originUrl);
  const importedIds = new Set(imported.map((channel) => channel.id));
  const current = loadDB();
  const kept = current.filter((item) => item.source !== "embedtv-channels" || importedIds.has(item.id));
  const byId = new Map(kept.map((item) => [String(item.id), item]));
  for (const channel of imported) byId.set(channel.id, { ...byId.get(channel.id), ...channel });
  saveDB([...byId.values()]);
  for (const channel of imported) {
    try {
      const existing = hls.channelById(channel.id);
      if (existing && existing.url !== channel.originUrl) hls.remove(channel.id);
      await hls.add({ id: channel.id, name: channel.name, url: channel.originUrl, automatic: false });
    }
    catch (error) { console.warn("[EMBEDTV] Canal não registrado no proxy:", channel.id, error.message); }
  }
  return { categories: payload.categories.length, imported: imported.length, updatedAt: new Date(channelsUpdatedAt).toISOString() };
}

async function resetChannels() {
  const current = loadDB();
  const embedChannels = current.filter((item) => item.source === "embedtv-channels");
  // Limpa também registros órfãos que ficaram apenas no hls-channels.json.
  // Sem isso, hls.add() recusa o mesmo ID quando a origem/token mudou.
  const proxyIds = hls.list()
    .map((channel) => String(channel.id || ""))
    .filter((id) => id.startsWith("embedtv-channel-"));
  for (const id of new Set([
    ...embedChannels.map((channel) => String(channel.id)),
    ...proxyIds,
  ])) hls.remove(id);
  saveDB(current.filter((item) => item.source !== "embedtv-channels"));
  channelsCache = null;
  channelsUpdatedAt = 0;
  const payload = await loadChannels(true);
  const result = await syncChannels(true);
  return { removed: embedChannels.length, ...result, apiChannels: payload.channels.length };
}

async function registerPlayers(events) {
  // A sincronização deve registrar qualquer evento com player válido.
  // O filtro de Brasileirão é aplicado somente quando solicitado pela rota,
  // não durante o registro interno dos HLS.
  const selected = Array.isArray(events) ? events : [];
  let registered = 0;
  let failed = 0;
  for (const event of selected) {
    const id = idFor(event);
    if (!id) continue;
    const start = Date.parse(event.time_start || "");
    if (Number.isFinite(start) && Date.now() >= start + 5 * 60 * 60 * 1000) continue;
    const player = (Array.isArray(event.players) ? event.players : [])
      .map(extractPlayer).find((url) => url && allowedEmbed(url));
    if (!player) continue;
    try {
      await hls.add({ id, name: event.title || id, url: player, automatic: true });
      registered++;
    } catch (error) {
      failed++;
      console.error("[EMBEDTV] Falha:", id, error.message);
    }
  }
  return { registered, failed };
}

function syncEventCatalog(eventsList) {
  const today = localDate(Date.now());
  const active = new Map(
    hls.list().filter(channel => channel.automatic).map(channel => [channel.id, channel])
  );
  const desired = new Map();

  for (const event of eventsList) {
    const id = idFor(event);
    const channel = active.get(id);
    if (!channel || !event.time_start || localDate(event.time_start) !== today) continue;
    const poster = event.teams?.home?.image || null;
    desired.set(id, {
      id,
      name: event.title || id,
      url: `${BASE_URL}/tvbox/api/live/${encodeURIComponent(id)}.m3u8`,
      type: "hls",
      category: EVENT_CATEGORY,
      categories: [EVENT_CATEGORY],
      poster,
      source: EVENT_SOURCE,
      sourceId: String(event.id || event.slug || id),
    });
  }

  const existing = loadDB();
  const preserved = existing.filter(item => item.source !== EVENT_SOURCE);
  const occupied = new Set(preserved.map(item => String(item.id)));
  const previous = new Map(
    existing.filter(item => item.source === EVENT_SOURCE).map(item => [String(item.id), item])
  );
  for (const [id, item] of desired) {
    if (occupied.has(id)) {
      console.warn("[EMBEDTV] ID ocupado no catálogo:", id);
      continue;
    }
    preserved.push({ ...previous.get(id), ...item });
  }
  saveDB(preserved);
  return { active: desired.size, removed: previous.size - [...desired.keys()].filter(id => previous.has(id)).length };
}

function publicIP(ip) {
  if (net.isIP(ip) !== 4) return false;
  const [a, b, c] = ip.split(".").map(Number);
  return !(
    a === 0 ||
    a === 10 ||
    a === 127 ||
    a >= 224 ||
    (a === 169 && b === 254) ||
    (a === 172 && b >= 16 && b <= 31) ||
    (a === 192 && b === 168) ||
    (a === 100 && b >= 64 && b <= 127) ||
    (a === 192 && b === 0) ||
    (a === 192 && b === 88 && c === 99) ||

async function validate(raw) {
  const u = new URL(raw);
  if (
    !["https:", "http:"].includes(u.protocol) ||
    u.username ||
    u.password ||
    (u.port && !["80", "443"].includes(u.port))
  )
    throw Error("URL não permitida");
  if (net.isIP(u.hostname)) {
    if (!publicIP(u.hostname)) throw Error("IP privado não permitido");
  } else {
    const ips = await dns.lookup(u.hostname, { all: true, family: 4 });
    if (!ips.length || ips.some((x) => !publicIP(x.address)))
      throw Error("Endereço de origem não permitido");
  }
  return u.href;
}
async function request(raw, headers = {}) {
  let url = raw;
  for (let i = 0; i < 5; i++) {
    url = await validate(url);
    const response = await fetch(url, {
      redirect: "manual",
      signal: AbortSignal.timeout(15000),
      headers: { "User-Agent": "Mozilla/5.0", Accept: "*/*", ...headers },
    });
    if (response.status >= 300 && response.status < 400) {
      const next = response.headers.get("location");
      await response.body?.cancel();
      if (!next) throw Error("Redirect inválido");
      url = new URL(next, url).href;
      continue;
    }
    return { response, url };
  }
  throw Error("Redirecionamentos demais");
}
async function textLimited(response) {
  const size = Number(response.headers.get("content-length") || 0);
  if (size > 2_000_000) throw Error("HTML/playlist grande demais");
  const data = await response.arrayBuffer();
  if (data.byteLength > 2_000_000) throw Error("HTML/playlist grande demais");
  return Buffer.from(data).toString("utf8");
}

// A página EmbedTV e o endereço temporário podem ter bloqueio/região por IP.
// A resolução usa o WARP quando HLS_USE_WARP não estiver explicitamente
// desativado. O fallback direto mantém o serviço funcionando caso o WARP
// esteja parado durante uma reinicialização.
async function requestEmbed(raw, headers = {}) {
  if (String(process.env.HLS_USE_WARP || "1") === "0")
    return request(raw, headers);
  try {
    let url = raw;
    for (let i = 0; i < 5; i++) {
      const target = await validate(url);
      const response = await axios.get(target, {
        httpAgent: warpAgent,
        httpsAgent: warpAgent,
        proxy: false,
        maxRedirects: 0,
        responseType: "arraybuffer",
        timeout: 15000,
        validateStatus: () => true,
        headers: { "User-Agent": "Mozilla/5.0", Accept: "*/*", ...headers },
      });
      if (
        response.status >= 300 &&
        response.status < 400 &&
        response.headers.location
      ) {
        url = new URL(response.headers.location, target).href;
        continue;
      }
      const body = Buffer.from(response.data || []);
      return {
        url: target,
        response: {
          ok: response.status >= 200 && response.status < 300,
          status: response.status,
          headers: {
            get: (name) => response.headers[String(name).toLowerCase()] || null,
          },
          arrayBuffer: async () => body,
        },
      };
    }
    throw Error("Redirecionamentos demais");
  } catch (error) {
    console.warn(
      `[HLS] WARP indisponível; usando conexão direta: ${error.message}`,
    );
    return request(raw, headers);
  }
}
function candidates(body, base) {
  const txt = body
    .replace(/\\u002f/gi, "/")
    .replace(/\\u003a/gi, ":")
    .replace(/\\u0026/gi, "&")
    .replace(/\\\//g, "/")
    .replace(/&amp;/g, "&")
    .replace(/&quot;/gi, '"')
    .replace(/&#x2f;/gi, "/");
  const found = new Set();
  for (const m of txt.matchAll(
    /(?:https?:\/\/[^\s"'<>]+|["']([^"']+)["'])/gi,
  )) {
    const candidate = (m[1] || m[0]).replace(/[\\'";,)>]+$/g, "");
    if (!/\.(?:m3u8|txt)(?:[?#]|$)/i.test(candidate)) continue;
    try {
      found.add(new URL(candidate, base).href);
    } catch {}
  }
  return [...found].slice(0, 30);
}
async function resolve(channel) {
  const root = await requestEmbed(channel.url);
  if (!root.response.ok) throw Error("Origem HTTP " + root.response.status);
  const body = await textLimited(root.response);
  if (body.trimStart().startsWith("#EXTM3U")) return { url: root.url };
  const urls = candidates(body, root.url);
  const rootCookie = root.response.headers.get("set-cookie");
  console.log(
    `[HLS] Embed ${channel.id || channel.url}: ${urls.length} candidato(s)`,
  );
  for (const candidate of urls) {
    try {
      const candidateHeaders = {
        Referer: root.url,
        Origin: new URL(root.url).origin,
        ...(rootCookie
          ? {
              Cookie: rootCookie
                .split(",")
                .map((v) => v.split(";")[0])
                .join("; "),
            }
          : {}),
      };
      let p = await requestEmbed(candidate, candidateHeaders);
      // Alguns hosts recusam o IP do WARP, mas aceitam a conexão direta.
      // Tenta a origem uma segunda vez antes de descartar o canal.
      if ([401, 403].includes(p.response.status)) {
        console.warn(
          `[HLS] ${channel.id || channel.url}: WARP recusado (${p.response.status}), tentando direto`,
        );
        p = await request(candidate, candidateHeaders);
      }
      if (!p.response.ok) {
        console.warn(
          `[HLS] ${channel.id || channel.url}: candidato HTTP ${p.response.status} ${candidate}`,
        );
        continue;
      }
      const t = await textLimited(p.response);
      if (t.trimStart().startsWith("#EXTM3U")) return { url: p.url };
      console.warn(
        `[HLS] ${channel.id || channel.url}: candidato não é playlist ${candidate}`,
      );
    } catch (e) {
      console.warn(
        `[HLS] ${channel.id || channel.url}: falha no candidato ${candidate}: ${e.message}`,
      );
    }
  }
  throw Error("Não foi encontrada playlist HLS no embed");
}

// Modo opcional para origens que bloqueiam o IP do servidor: o Node apenas
// descobre o endereço temporário (`error.txt`/`.m3u8`) e o cliente acessa a
// playlist diretamente. O token fica exposto ao cliente e pode expirar.
async function resolveRedirect(channel) {
  const root = await requestEmbed(channel.url);
  if (!root.response.ok) throw Error("Origem HTTP " + root.response.status);
  const body = await textLimited(root.response);
  if (body.trimStart().startsWith("#EXTM3U")) return root.url;
  const urls = candidates(body, root.url);
  if (!urls.length) throw Error("Não foi encontrada playlist HLS no embed");
  return urls[0];
}

async function refresh(channel, force = false) {
  const old = cache.get(channel.id);
  if (old?.pending) return old.pending;
  // A URL HLS extraída de um embed pode ter token temporário. Ela fica em
  // cache normalmente, mas um refresh forçado deve consultar o embed de novo.
  if (channel.automatic && old?.url && !force) return old;
  if (!force && old?.checked) {
    // Após falha (especialmente 403), aguarda mais tempo antes de tentar de
    // novo. Repetir a requisição a cada ciclo só aumenta o bloqueio no host.
    const waitMs = old.error
      ? Math.max(settings.pollingSeconds * 2 * 1000, 10 * 60 * 1000)
      : settings.pollingSeconds * 1000;
    if (Date.now() - old.checked < waitMs) return old;
  }
  const promise = (async () => {
    try {
      const found = await resolve(channel);
      const value = {
        url: found.url,
        checked: Date.now(),
        error: null,
        checks: (old?.checks || 0) + 1,
        changes:
          (old?.changes || 0) + (old?.url && old.url !== found.url ? 1 : 0),
      };
      cache.set(channel.id, value);
      return value;
    } catch (e) {
      const value = {
        ...old,
        checked: Date.now(),
        error: e.message,
        checks: (old?.checks || 0) + 1,
      };
      cache.set(channel.id, value);
      return value;
    }
  })();
  cache.set(channel.id, { ...old, pending: promise });
  return promise;
}

function cleanup() {
  let changed = false;
  for (const [id, channel] of channels) {
    if (
      channel.automatic &&
      (!channel.expiresAt || channel.expiresAt <= Date.now())
    ) {
      channels.delete(id);
      cache.delete(id);
      changed = true;
    }
  }
  for (const [token, resource] of resources) {
    if (
      !channels.has(resource.channelId) ||
      Date.now() - resource.lastAccess > RESOURCE_IDLE_TTL_MS
    )
      resources.delete(token);
  }
  if (changed) persist();
}

async function add({ name, url, id, automatic = false }) {
  name = String(name || "").trim();
  url = String(url || "").trim();
  id = slug(id || name);

  if (!name || name.length > 100 || !id || !url || url.length > 4096) {
    throw Error("Canal inválido");
  }

  cleanup();
  await validate(url);

  const existing = channels.get(id);

  if (existing) {
    if (existing.automatic && !automatic) {
      existing.automatic = false;
      delete existing.createdAt;
      delete existing.expiresAt;
      channels.set(id, existing);
      persist();
    }
    // A fonte do mesmo evento pode mudar. Preserve o cache da sessão atual e
    // use a nova fonte somente quando a playlist precisar ser renovada.
    if (existing.automatic && automatic) {
      existing.name = name;
      existing.url = url;
      existing.createdAt = existing.createdAt || Date.now();
      existing.expiresAt = Date.now() + AUTO_TTL_MS;
      channels.set(id, existing);
      persist();
      return existing;
    }
    if (existing.url !== url) {
      throw Error("ID já usado para outra URL");
    }

    // O polling periódico já atualiza canais existentes.
    // Evita disparar várias verificações simultâneas.
    return existing;
  }

  const channel = {
    id,
    name,
    url,
    automatic,
    ...(automatic
      ? { createdAt: Date.now(), expiresAt: Date.now() + AUTO_TTL_MS }
      : {}),
  };

  channels.set(id, channel);
  persist();

  console.log(`[HLS] Canal cadastrado: ${id}`);

  // Não bloqueia o cadastro aguardando a origem.
  refresh(channel, true).catch((error) => {
    console.error(`[HLS] Erro ao verificar ${id}:`, error.message);
  });

  return channel;
}

}
function removeAll() {
  const removed = channels.size;

  cache.clear();
  resources.clear();
  persist();
  return removed;
}

function tokenFor(channelId, url) {
  const token = crypto
    .createHash("sha256")
    .update(channelId + "\0" + url)
    .digest("hex")
    .slice(0, 32);
  const now = Date.now();
  const previous = resources.get(token);
  resources.set(token, {
    channelId,
    url,
    created: previous?.created || now,
    lastAccess: now,
  });
  return token;
}

function rewrite(body, base, id) {
  return body
    .split(/\r?\n/)
    .map((line) => {
      if (!line.trim()) return line;

      if (line.trimStart().startsWith("#")) {
        return line.replace(/URI="([^"]+)"/g, (_, relative) => {
          const absolute = new URL(relative, base).href;

          const token = tokenFor(id, absolute);

          return `URI="/tvbox/api/live/resource/${token}"`;
        });
      }

      const absolute = new URL(line.trim(), base).href;

      const token = tokenFor(id, absolute);

      return `/tvbox/api/live/resource/${token}`;
    })
    .join("\n");
}

async function play(req, res) {
  cleanup();
  const c = channels.get(req.params.id);
  if (!c) return res.status(404).send("Canal não encontrado");
  try {
    if (
      String(process.env.HLS_DIRECT_REDIRECT || "0") === "1" &&
      /(^|\.)embedtv\.cc$/i.test(new URL(c.url).hostname)
    ) {
      const target = await resolveRedirect(c);
      return res.redirect(302, target);
    }
    let state = await refresh(c);
    if (!state.url) throw Error(state.error || "Origem indisponível");
    // Playlist sempre nova, nunca guardar segmentos ao longo do polling.
    // A playlist temporária também precisa ser buscada pelo mesmo caminho
    // (WARP + Referer do embed); consultar direto costuma retornar 403.
    const sourceHeaders = {
      Referer: c.url,
      Origin: new URL(c.url).origin,
    };
    let p = await requestEmbed(state.url, sourceHeaders);
    if (!p.response.ok && [401, 403, 404, 410].includes(p.response.status)) {
      await p.response.body?.cancel();
      state = await refresh(c, true);
      if (!state.url)
        throw Error(state.error || "Unable to renew stream source");
      p = await requestEmbed(state.url, sourceHeaders);
    }
    if (!p.response.ok) throw Error("Origem HTTP " + p.response.status);
    const body = await textLimited(p.response);
    if (!body.trimStart().startsWith("#EXTM3U"))
      throw Error("Playlist inválida");
    res.set({
      "Content-Type": "application/vnd.apple.mpegurl",
      "Cache-Control": "no-store",
      "Access-Control-Allow-Origin": "*",
    });
    return res.send(rewrite(body, p.url, c.id));
  } catch (e) {
    return res.status(502).send(e.message);
  }
}
async function resource(req, res) {
  cleanup();
  const entry = resources.get(req.params.token);
  if (!entry || !channels.has(entry.channelId))
    return res.status(404).send("Recurso expirado");
  entry.lastAccess = Date.now();
  try {
    const { response, url } = await request(
      entry.url,
      req.headers.range ? { Range: req.headers.range } : {},
    );
    if (!response.ok && response.status !== 206)
      throw Error("Origem HTTP " + response.status);
    const type = response.headers.get("content-type") || "";
    const isPlaylist =
      /mpegurl/i.test(type) || /\.(m3u8|txt)$/i.test(new URL(url).pathname);
    if (isPlaylist) {
      const buf = await response.arrayBuffer();
      const body = Buffer.from(buf).toString("utf8");
      if (body.trimStart().startsWith("#EXTM3U")) {
        res.set({
          "Content-Type": "application/vnd.apple.mpegurl",
          "Cache-Control": "no-store",
          "Access-Control-Allow-Origin": "*",
        });
        return res.send(rewrite(body, url, entry.channelId));
      }
      if (/mpegurl/i.test(type)) throw Error("Playlist inválida");
      res.status(response.status).set({
        "Content-Type": type || "application/octet-stream",
        "Cache-Control": "no-store",
        "Access-Control-Allow-Origin": "*",
      });
      return res.send(Buffer.from(buf));
    }
    res.status(response.status).set({
      "Content-Type": type || "application/octet-stream",
      "Cache-Control": "no-store",
      "Access-Control-Allow-Origin": "*",
    });
    if (response.headers.get("content-range"))
      res.set("Content-Range", response.headers.get("content-range"));
    if (response.headers.get("accept-ranges"))
      res.set("Accept-Ranges", response.headers.get("accept-ranges"));
    await pipeline(Readable.fromWeb(response.body), res);
  } catch (e) {
    if (!res.headersSent) res.status(502).send(e.message);
    else res.destroy();
  }
}

  return { ...settings, lastRun, running, channels: channels.size };
}
function setSettings(next) {
  const n = Number(next.pollingSeconds);
  if (!Number.isInteger(n) || n < 30 || n > 3600)
    throw Error("Polling deve ser de 30 a 3600 segundos");
  settings.pollingSeconds = n;
  fs.writeFileSync(SETTINGS, JSON.stringify(settings, null, 2));

}
async function poll() {
  cleanup();
  if (running) return;
  running = true;
  try {
    // Não força uma nova descoberta em todos os canais. O cache e o backoff
    // controlam a frequência; canais em reprodução são renovados por play().
    const list = [...channels.values()]
      .filter((channel) => !channel.automatic)
      .sort(
        (a, b) =>
          (cache.get(a.id)?.checked || 0) - (cache.get(b.id)?.checked || 0),
      )
      .slice(0, MAX_REFRESH_PER_RUN);
    let i = 0;
    await Promise.all(
      Array.from(
        { length: Math.min(POLL_CONCURRENCY, list.length) },
        async () => {
          while (i < list.length) {
            const channel = list[i++];
            await refresh(channel, false);
          }
        },
      ),
    );
    lastRun = new Date().toISOString();
  } finally {
    running = false;
  }
}
function start() {
  if (interval) clearInterval(interval);
  interval = setInterval(
    () => poll().catch((e) => console.warn("[HLS]", e.message)),
    settings.pollingSeconds * 1000,
  );
  interval.unref?.();
}

        },
      );

        source: "IMDb",
      };

  }

}

function convertTo24h(timeStr, countdownText) {
  if (!timeStr) return "";
  if (countdownText && countdownText.toUpperCase().includes("LIVE")) return "AO VIVO";

  const match = timeStr.match(/(\d{1,2}):(\d{2})/);
  if (!match) return timeStr;

  let hours = parseInt(match[1], 10);
  const minutes = match[2];

  const upper = timeStr.toUpperCase();
  if (upper.includes("PM") && hours < 12) {
    hours += 12;
  } else if (upper.includes("AM") && hours === 12) {
    hours = 0;
  }

  hours = (hours + 4) % 24;

  const formattedHours = String(hours).padStart(2, "0");
  return timeStr.replace(/(\d{1,2}):(\d{2})/, `${formattedHours}:${minutes}`);
}

const crypto = require("crypto");

const SESSION_COOKIE = "tvbox_session";

function safeText(value = "") {
  return String(value).replace(
    /[&<>"']/g,
    (char) =>
      ({
        "&": "&amp;",
        "<": "&lt;",
        ">": "&gt;",
        '"': "&quot;",
        "'": "&#039;",
      })[char]
  );
}

function readCookies(req) {
  return Object.fromEntries(
    String(req.headers.cookie || "")
      .split(";")
      .map((part) => part.trim().split(/=(.*)/s, 2))
      .filter(([key]) => key)
      .map(([key, value]) => [key, decodeURIComponent(value || "")]),
  );
}

function signSession(payload) {
  const encoded = Buffer.from(JSON.stringify(payload)).toString("base64url");
  const signature = crypto
    .createHmac("sha256", SESSION_SECRET)
    .update(encoded)
    .digest("base64url");
  return `${encoded}.${signature}`;
}

function hasTvboxSession(req) {
  if (!SESSION_SECRET) return false;
  const token = readCookies(req)[SESSION_COOKIE];
  if (!token || !token.includes(".")) return false;
  const [encoded, signature] = token.split(".");
  const expected = crypto.createHmac("sha256", SESSION_SECRET).update(encoded).digest("base64url");
  if (signature.length !== expected.length || !crypto.timingSafeEqual(Buffer.from(signature), Buffer.from(expected))) return false;
  try {
    return JSON.parse(Buffer.from(encoded, "base64url").toString("utf8")).expiresAt > Date.now();
  } catch (_) {
    return false;
  }
}

function requireTvboxAccess(req, res) {
  if (hasTvboxSession(req)) return true;
  if (req.accepts(["html"])) res.redirect("/tvbox/login");
  else res.status(401).json({ error: "Sessão expirada. Faça login novamente." });
  return false;
}

function clearSession(res) {
  res.clearCookie(SESSION_COOKIE, { path: "/tvbox", sameSite: "lax" });
}

function passwordMatches(candidate) {
  if (!PASSWORD) return false;
  const received = Buffer.from(String(candidate || ""));
  const expected = Buffer.from(PASSWORD);
  return received.length === expected.length && crypto.timingSafeEqual(received, expected);
}

function removeLegacyPasswordFromHtml(html) {
  // Esta função também recebe páginas com JavaScript embutido. Não tente
  // normalizar `?` no HTML inteiro: isso altera operadores ternários do JS
  // (por exemplo, `condicao ? "a" : "b"`) e quebra a página no navegador.
  // A remoção do parâmetro legado continua limitada ao padrão `pass=`.
  return html.replace(/([?&])pass=[^&"'\s>]+&?/g, "$1");
}

<script>
(function () {
  "use strict";

  const $ = function (id) {
    return document.getElementById(id);
  };

  const BASE = "/tvbox/hls";

  let loading = false;

  function message(text, error) {
    const el = $("message");

    if (!el) return;

    el.textContent = String(text || "");

    if (error === true) {
      el.style.color = "#f87171";
    } else {
      el.style.color = "";
    }
  }

  function formatError(error) {
    if (error && error.message) {
      return error.message;
    }

    return String(error);
  }

  async function api(endpoint, options) {
    const url = BASE + endpoint;
    const config = options || {};

    const headers = {
      Accept: "application/json"
    };

    if (config.body !== undefined) {
      headers["Content-Type"] = "application/json";
    }

    if (config.headers) {
      Object.assign(headers, config.headers);
    }

    let response;

    try {
      response = await fetch(url, {
        method: config.method || "GET",
        credentials: "same-origin",
        cache: "no-store",
        headers: headers,
        body: config.body
      });
    } catch (error) {
      throw new Error("Falha de conexão com " + url + ": " + formatError(error));
    }

    if (response.status === 401) {
      throw new Error("Sessão expirada. Faça login novamente em /tvbox/login");
    }

    const raw = await response.text();

    let data;

    try {
      data = JSON.parse(raw);
    } catch (error) {
      if (
        response.redirected ||
        raw.toLowerCase().includes("<html") ||
        raw.toLowerCase().includes("<!doctype")
      ) {
        throw new Error("A rota " + url + " retornou HTML em vez de JSON. Verifique sua sessão em /tvbox/login.");
      }

      throw new Error("Resposta inválida da API " + url + " (HTTP " + response.status + "): " + raw.substring(0, 200));
    }

    if (!response.ok) {
      throw new Error(url + " - HTTP " + response.status + ": " + (data.error || "Erro desconhecido"));
    }

    return data;
  }

  function createButton(label, action, danger) {
    const btn = document.createElement("button");

    btn.type = "button";
    btn.className = danger ? "btn danger" : "btn";
    btn.textContent = label;

    btn.addEventListener("click", async function () {
      btn.disabled = true;

      try {
        await action();
      } catch (error) {
        message(formatError(error), true);
      } finally {
        btn.disabled = false;
      }
    });

    return btn;
  }

  function formatDate(value) {
    if (!value) return "-";

    const date = new Date(value);

    if (Number.isNaN(date.getTime())) {
      return String(value);
    }

    return date.toLocaleString("pt-BR");
  }

  async function loadSettings() {
    try {
      const config = await api("/settings");

      if (document.activeElement !== $("poll")) {
        $("poll").value = config.pollingSeconds || 300;
      }

      const status = $("status");

      status.style.color = "";

      status.textContent =
        "Polling: " +
        (config.pollingSeconds || 300) +
        " segundos | Canais: " +
        (config.channels || 0) +
        " | Última execução: " +
        formatDate(config.lastRun) +
        " | Executando: " +
        (config.running ? "Sim" : "Não");

    } catch (error) {
      $("status").textContent = "Erro ao carregar configurações: " + formatError(error);
      $("status").style.color = "#f87171";
    }
  }

  function createChannelCard(channel) {
    const card = document.createElement("article");

    card.className = "card";
    card.style.marginBottom = "14px";

    const title = document.createElement("h3");

    title.textContent = String(channel.name || "Sem nome") + " (" + String(channel.id || "-") + ")";

    const state = document.createElement("p");

    state.className = "muted";

    state.textContent =
      "Status: " +
      (channel.status || "checking") +
      " | Verificações: " +
      (channel.checks || 0) +
      " | Mudanças de URL: " +
      (channel.changes || 0) +
      " | Última atualização: " +
      formatDate(channel.lastCheck);

    const errorElement = document.createElement("p");

    if (channel.error) {
      errorElement.textContent = "Erro: " + channel.error;
      errorElement.style.color = "#f87171";
    }

    const urlElement = document.createElement("p");

    urlElement.className = "url";

    const publicUrl = new URL(
      "/tvbox/api/live/" + encodeURIComponent(channel.id) + ".m3u8",
      window.location.origin
    ).href;

    urlElement.textContent = publicUrl;

    const actions = document.createElement("div");

    actions.className = "actions";

    actions.appendChild(
      createButton("Copiar API", async function () {
        try {
          await navigator.clipboard.writeText(publicUrl);
          message("URL copiada: " + publicUrl);
        } catch (error) {
          window.prompt("Copie a URL do canal:", publicUrl);
        }
      })
    );

    actions.appendChild(
      createButton("Abrir playlist", async function () {
        window.open(publicUrl, "_blank", "noopener");
      })
    );

    actions.appendChild(
      createButton("Atualizar agora", async function () {
        message("Atualizando canal " + channel.name + "...");

        await api(
          "/channels/" + encodeURIComponent(channel.id) + "/refresh",
          {
            method: "POST"
          }
        );

        message("Verificação concluída: " + channel.name);

        await loadChannels();
      })
    );

    actions.appendChild(
      createButton("Excluir", async function () {
        const confirmed = window.confirm("Deseja excluir o canal " + channel.name + "?");

        if (!confirmed) return;

        await api(
          "/channels/" + encodeURIComponent(channel.id),
          {
            method: "DELETE"
          }
        );

        message("Canal excluído com sucesso.");

        await loadChannels();
        await loadSettings();
      }, true)
    );

    card.appendChild(title);
    card.appendChild(state);

    if (channel.error) {
      card.appendChild(errorElement);
    }

    card.appendChild(urlElement);
    card.appendChild(actions);

    return card;
  }

  async function loadChannels() {
    if (loading) return;

    loading = true;

    try {
      const channels = await api("/channels");

      if (!Array.isArray(channels)) {
        throw new Error("A API /tvbox/hls/channels não retornou uma lista.");
      }

      const container = $("channels");

      container.replaceChildren();

      $("channelCount").textContent = channels.length + " canal(is) cadastrado(s)";

      if (channels.length === 0) {
        container.textContent = "Nenhum canal cadastrado.";
        return;
      }

      channels.forEach(function (channel) {
        container.appendChild(createChannelCard(channel));
      });

    } catch (error) {
      $("channels").textContent = "Erro ao carregar canais: " + formatError(error);
      message(formatError(error), true);
    } finally {
      loading = false;
    }
  }

  $("savePoll").addEventListener("click", async function () {
    const button = $("savePoll");

    button.disabled = true;

    try {
      const seconds = Number($("poll").value);

      if (!Number.isInteger(seconds) || seconds < 30 || seconds > 3600) {
        throw new Error("O intervalo deve estar entre 30 e 3600 segundos.");
      }

      await api("/settings", {
        method: "POST",
        body: JSON.stringify({
          pollingSeconds: seconds
        })
      });

      message("Intervalo salvo com sucesso.");

      await loadSettings();

    } catch (error) {
      message(formatError(error), true);
    } finally {
